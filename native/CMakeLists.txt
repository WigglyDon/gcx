# Mechanical adapter over Dolphin; the product and host loop are Rust.
add_library(ntx_engine SHARED bridge.cpp)
if(NOT ANDROID)
  target_sources(ntx_engine PRIVATE "${PROJECT_SOURCE_DIR}/Source/Core/InputCommon/ControllerInterface/Touch/InputOverrider.cpp")
  target_sources(ntx_engine PRIVATE host_fixture.cpp)
endif()
target_compile_features(ntx_engine PRIVATE cxx_std_23)
target_include_directories(ntx_engine PRIVATE "${PROJECT_SOURCE_DIR}/Source/Core")
target_link_libraries(ntx_engine PRIVATE core uicommon imgui)
target_compile_options(ntx_engine PRIVATE -Wall -Wextra -Werror -Wno-unused-parameter)
if(ANDROID)
  add_library(androidcommon STATIC android_platform.cpp)
  target_compile_features(androidcommon PRIVATE cxx_std_23)
  target_include_directories(androidcommon PRIVATE "${PROJECT_SOURCE_DIR}/Source/Android" "${PROJECT_SOURCE_DIR}/Source/Core")
  target_link_libraries(androidcommon PRIVATE common android log)
  target_link_libraries(ntx_engine PRIVATE android log)
  target_link_options(ntx_engine PRIVATE "-Wl,-z,max-page-size=16384")
endif()
set_target_properties(ntx_engine PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/ntx")
if(UNIX AND NOT APPLE)
  target_link_options(ntx_engine PRIVATE "-Wl,--no-undefined")
endif()
